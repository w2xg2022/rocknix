setenv rocknix_kernel_addr "0x02080000"
setenv rocknix_fdt_addr "0x08300000"
if test -e ${devtype} ${devnum} rocknix/TRIGGER; then
	echo "=== ROCKNIX chainload triggered ==="
	load ${devtype} ${devnum} ${rocknix_kernel_addr} rocknix/KERNEL
	load ${devtype} ${devnum} ${rocknix_fdt_addr} rocknix/dtb
	setenv bootargs "boot=LABEL=ROCKNIX disk=LABEL=STORAGE quiet console=ttyS2,1500000"
	booti ${rocknix_kernel_addr} - ${rocknix_fdt_addr}
	echo "=== ROCKNIX booti failed, falling back to Armbian ==="
fi
